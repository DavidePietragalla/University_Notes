transformation Families2Persons(families: Families, persons: Persons) {

    // Rule 1: Transform a father into a Male object
    top relation FatherToMale {
        
        // Variable declarations for pattern matching
        fname, lname: String;
        
        // Source domain: match families that have a defined father
        checkonly domain families f:Family {
            
            // Bind the family's last name to the 'lname' variable
            lastName = lname,
            
            // Navigate the 'father' reference to access the Member object
            father = m:Member {
                
                // Bind the father's first name to the 'fname' variable
                firstName = fname
            }
        };

        // Target domain: enforce the creation of a Male object with the concatenated full name
        enforce domain persons p:Male {
            fullName = fname + ' ' + lname
        };
    }
    
    // Rule 2: Transform a mother into a Female object
    top relation MotherToFemale {
        
        fname, lname: String;
        
        checkonly domain families f:Family {
            
            lastName = lname,
            
            mother = m:Member {
                
                firstName = fname
            }
        };

        enforce domain persons p:Female {
            fullName = fname + ' ' + lname
        };

    }

    // Rule 3: Transform a son into a Male object
    top relation SonToMale {
        
        fname, lname: String;
        
        checkonly domain families f:Family {
            
            lastName = lname,
            
            sons = m:Member {
                
                firstName = fname
            }
        };

        enforce domain persons p:Male {
            fullName = fname + ' ' + lname
        };

    }

    // Rule 4: Transform a daughter into a Female object
    top relation DaughterToFemale {
        
        fname, lname: String;
        
        checkonly domain families f:Family {
            
            lastName = lname,
            
            daughters = m:Member {
                
                firstName = fname
            }
        };

        enforce domain persons p:Female {
            fullName = fname + ' ' + lname
        };

    }

}
